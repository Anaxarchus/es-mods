# The data half of the "Diplomacy" plugin. Its Dado mod (scripts/diplomacy.js)
# runs the wars; this file defines the governments its flotillas fly under, the
# faction missions offered to members, and the Luna Navy yard event. The rebel
# governments and their fleets are in "diplomacy factions.txt", the spaceport
# news in "diplomacy news.txt".
#
# The flotilla governments look like their faction but are separate, so a war
# is only hostility between flotilla governments (which the mod sets with
# event changes) and the game's own politics are never touched.

government "Republic Flotilla"
	"display name" "Republic"
	swizzle 0
	color "governments: Republic"
	"player reputation" 1
	"attitude toward"
		"Republic" 1
		"Merchant" .3
		"Militia" .3
		"Pirate" -.4

government "Free Worlds Flotilla"

# Faction missions: offered on the job board to members of a faction at war.
# The mod sets "diplomacy: member" (1 Republic, 2 Free Worlds, 3 Syndicate)
# and "diplomacy: at war <faction>"; completing one adds to "diplomacy: merit",
# which the mod turns into rank and salary, and to "diplomacy: dispatches",
# which it turns into relief from war exhaustion for the member's faction.

mission "Diplomacy: Republic Dispatches"
	name "Dispatches to <planet>"
	job
	repeat
	description "Carry sealed war dispatches for the Republic Navy to <destination> by <date>. Payment is <payment>, and the run counts toward your merit."
	deadline
	cargo "war dispatches" 1
	to offer
		"diplomacy: member" == 1
		"diplomacy: at war 1" == 1
	source
		government "Republic"
	destination
		distance 2 5
		government "Republic"
	on complete
		payment 20000 150
		"diplomacy: merit" += 5
		"diplomacy: dispatches" += 1
		dialog `A Navy officer signs for the dispatches and pays you <payment>. "Good work, Captain. The fleet won't forget it."`

# The Republic's capital-warship yard. The Free Worlds campaign wrecks Geminus
# (event "war begins" clears its shipyard), the only world that sells the
# Navy Advanced hulls (Frigate, Cruiser, Carrier, Auxiliary), until "geminus
# rebuilt" 90 days after the campaign; and in Diplomacy Geminus (Castor, in
# the Paradise Worlds) can change hands. Luna, in Sol, which is never taken,
# sells them too. A planet's "shipyard" line in an event adds to its yards,
# so Luna keeps its own. The Free Worlds Veteran start applies this event; for
# any other pilot the mod applies the same change once (condition "diplomacy:
# luna yard") when the campaign is over or Geminus lies wrecked.
event "diplomacy: luna navy yard"
	planet "Luna"
		shipyard "Navy Advanced"
