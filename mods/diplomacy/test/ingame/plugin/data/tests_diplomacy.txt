test "Diplomacy - War Begins"
	status active
	description "The mod activates on its own, the forced war starts on the first day, and the Free Worlds hold their capital."
	sequence
		inject "Diplomacy Test Save"
		call "Load First Savegame"
		assert
			"diplomacy: active" == 1
			"diplomacy: war a" == 0
		call "Depart"
		apply
			"test: steps to wait" = 30
		label "flying"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "flying"
			"test: steps to wait" > 0
		assert
			"mod loaders" >= 1
			"diplomacy: war a" == 1
			"diplomacy: war b" == 2
			"diplomacy: at war 1" == 1
			"diplomacy: at war 2" == 1
			"diplomacy: stat wars" == 1
			"diplomacy: last day" > 0


test "Diplomacy - Flotilla In Person"
	status active
	description "A flotilla launched in the player's system is there as real ships when the player arrives, and the war goes on day by day."
	sequence
		inject "Diplomacy Test Save"
		call "Load First Savegame"
		navigate
			travel Vega
		call "Depart"
		# The takeoff's day starts the war.
		label "departing"
		input
		branch "departing"
			"diplomacy: war a" == 0
		# The jump's day puts a Free Worlds flotilla where the player arrives.
		apply
			"diplomacy: force flotilla" = 2
		input
			command jump
		label "toVega"
		branch "toVega"
			not "flagship system: Vega"
		apply
			"test: steps to wait" = 3
		label "watching"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "watching"
			"test: steps to wait" > 0
		assert
			"diplomacy: war a" == 1
			"diplomacy: ships here" > 0
			"diplomacy: stat launches" >= 1
		# Keep jumping, one leg at a time: days pass and the flotillas move on.
		navigate
			travel Sol
		input
			command jump
		label "toSol"
		branch "toSol"
			not "flagship system: Sol"
		navigate
			travel Vega
		input
			command jump
		label "toVega2"
		branch "toVega2"
			not "flagship system: Vega"
		navigate
			travel Sol
		input
			command jump
		label "toSol2"
		branch "toSol2"
			not "flagship system: Sol"
		navigate
			travel Vega
		input
			command jump
		label "toVega3"
		branch "toVega3"
			not "flagship system: Vega"
		assert
			"diplomacy: last day" >= "diplomacy: activated" + 5
			"diplomacy: war a" == 1


test "Diplomacy - Join And Leave"
	status active
	description "On a Republic world the planet screen's Faction button opens the faction panel; Join makes the player a member with a salary and the flotillas' trust, and Leave undoes it."
	sequence
		inject "Diplomacy Test Save"
		call "Load First Savegame"
		assert
			"diplomacy: member" == 0
		# Open the faction panel (the planet button's key) and press its first button, Join.
		input
			key k
		input
			key 1
		assert
			"diplomacy: member" == 1
			"salary: Republic commission" > 0
			"reputation: Republic Flotilla" > 0
		# The panel now offers Leave first.
		input
			key 1
		assert
			"diplomacy: member" == 0
			"salary: Republic commission" == 0
		input
			key escape

test "Diplomacy - Licenses And Jobs"
	status active
	description "A Republic member's rank issues the Navy licenses; the mod publishes the faction at war with the member's; and each kind of faction job against the Free Worlds finds its target or waypoints from a border world."
	sequence
		inject "Diplomacy Border Save"
		call "Load First Savegame"
		assert
			"diplomacy: member" == 1
			"diplomacy: rank" == 6
			"license: Navy" == 1
			"license: Navy Auxiliary" == 1
			"license: Navy Cruiser" == 1
			"license: Navy Carrier" == 1
		# Take off and land again: the test copies of the jobs are offered on
		# landing, and accepted at once since they have no conversation. A job
		# that cannot place its target or waypoints is not offered at all.
		# The forced war starts with the first day that passes, on taking off.
		call "Depart"
		apply
			"test: steps to wait" = 10
		label "flying"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "flying"
			"test: steps to wait" > 0
		assert
			"diplomacy: enemy" == 2
		call "Land"
		assert
			"Diplomacy Test: Republic Job: Courier vs Free Worlds: active" == 1
			"Diplomacy Test: Republic Job: Scout Ship vs Free Worlds: active" == 1
			"Diplomacy Test: Republic Job: Scout System vs Free Worlds: active" == 1
			"Diplomacy Test: Republic Job: Medium Warship vs Free Worlds: active" == 1
			"Diplomacy Test: Republic Job: Large Warship vs Free Worlds: active" == 1
			"Diplomacy Test: Republic Job: Warfleet vs Free Worlds: active" == 1
