test "Diplomacy - War Begins"
	status active
	description "The mod activates on its own, the forced war starts on the first day, and the Free Worlds hold their capital."
	sequence
		inject "Diplomacy Test Save"
		call "Load First Savegame"
		assert
			"diplomacy: active" == 1
			"diplomacy: war a" == 0
		call "Depart"
		apply
			"test: steps to wait" = 30
		label "flying"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "flying"
			"test: steps to wait" > 0
		assert
			"mod loaders" >= 1
			"diplomacy: war a" == 1
			"diplomacy: war b" == 2
			"diplomacy: at war 1" == 1
			"diplomacy: at war 2" == 1
			"diplomacy: stat wars" == 1
			"diplomacy: last day" > 0


test "Diplomacy - Flotilla In Person"
	status active
	description "A flotilla launched in the player's system is there as real ships when the player arrives, and the war goes on day by day."
	sequence
		inject "Diplomacy Test Save"
		call "Load First Savegame"
		navigate
			travel Vega
		call "Depart"
		# The takeoff's day starts the war.
		label "departing"
		input
		branch "departing"
			"diplomacy: war a" == 0
		# The jump's day puts a Free Worlds flotilla where the player arrives.
		apply
			"diplomacy: force flotilla" = 2
		input
			command jump
		label "toVega"
		branch "toVega"
			not "flagship system: Vega"
		apply
			"test: steps to wait" = 3
		label "watching"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "watching"
			"test: steps to wait" > 0
		assert
			"diplomacy: war a" == 1
			"diplomacy: ships here" > 0
			"diplomacy: stat launches" >= 1
		# Keep jumping, one leg at a time: days pass and the flotillas move on.
		navigate
			travel Sol
		input
			command jump
		label "toSol"
		branch "toSol"
			not "flagship system: Sol"
		navigate
			travel Vega
		input
			command jump
		label "toVega2"
		branch "toVega2"
			not "flagship system: Vega"
		navigate
			travel Sol
		input
			command jump
		label "toSol2"
		branch "toSol2"
			not "flagship system: Sol"
		navigate
			travel Vega
		input
			command jump
		label "toVega3"
		branch "toVega3"
			not "flagship system: Vega"
		assert
			"diplomacy: last day" >= "diplomacy: activated" + 5
			"diplomacy: war a" == 1


test "Diplomacy - Join And Leave"
	status active
	description "On a Republic world the planet screen's Faction button opens the faction panel; Join makes the player a member with a salary and the flotillas' trust, and Leave undoes it."
	sequence
		inject "Diplomacy Test Save"
		call "Load First Savegame"
		assert
			"diplomacy: member" == 0
		# Open the faction panel (the planet button's key) and press its first button, Join.
		input
			key k
		input
			key 1
		assert
			"diplomacy: member" == 1
			"salary: Republic commission" > 0
			"reputation: Republic Flotilla" > 0
		# The panel now offers Leave first.
		input
			key 1
		assert
			"diplomacy: member" == 0
			"salary: Republic commission" == 0
		input
			key escape

# A brand-new pilot on the "Free Worlds Veteran" start (data/diplomacy fw veteran.txt): the galaxy and
# conditions of a finished Free Worlds campaign, and the Diplomacy offer on Earth, which makes the pilot a
# Free Worlds Commodore.
test "Diplomacy - Veteran Start"
	status active
	description "A new pilot on the Free Worlds Veteran start begins after the campaign, with its events applied and scheduled and its conditions set, and is offered Diplomacy on landing; turning it on makes them a Free Worlds Commodore."
	sequence
		# Main menu: the pilots panel, then New Pilot, which opens the list of starts.
		input
			key l
		input
			key n
		# The Free Worlds Veteran is the last start in the list: wrap around to it.
		input
			key up
		input
			key s
		# The intro conversation applies the campaign's events, then asks for a name.
		# Typed too soon, the name is lost and the conversation waits: type it
		# again until the pilot has landed on Earth (its landing missions offered).
		apply
			"test: naming" = 0
		label "naming"
		input
		input
		input
			key a
		input
			key "return"
		input
			key b
		input
			key "return"
		input
			key "return"
		input
		input
		input
		apply
			"test: naming" = "test: naming" + 1
		branch "naming"
			"Main Plot Completed Date: offered" == 0
			"test: naming" < 10
		assert
			"start: diplomacy free worlds veteran" == 1
			"free worlds plot completed" == 1
			"main plot completed" == 1
			"free worlds reconciliation" == 1
			"chosen sides" == 1
			"license: Militia" == 1
			"salary: Free Worlds" == 8900
			"karma" == 3
			"FW Syndicate Extremists 1C: done" == 1
		# The campaign's events: the war, the territory, the armistice, the Pug, Algenib.
		assert
			"event: war begins" == 1
			"event: fw southern expansion" == 1
			"event: fw northern expansion" == 1
			"event: fw armistice" == 1
			"event: fw tarazed joins" == 1
			"event: navy out of rastaban" == 1
			"event: pug territory liberated" == 1
			"event: battle of algenib" == 1
		# Those still pending at the end are scheduled at their remaining offsets.
		assert
			"scheduled event: stack core for sale" == 1
			"scheduled event: pug link restoration 4" == 48
			"scheduled event: Greenrock recovered" == 121
		# Reputations, and the landing missions a finisher gets next (Pug Cease Hostilities makes peace with the Pug).
		assert
			"reputation: Republic" == 1
			"reputation: Syndicate" == 1
			"reputation: Pug" >= 1
			"Pug Cease Hostilities: offered" == 1
			"Main Plot Completed Date: offered" == 1
		# The ship and the rewards.
		assert
			"ship model: Osprey" == 1
			"outfit (flagship installed): Jump Drive" == 1
			"outfit (flagship installed): Hyperdrive" == 0
			"outfit: Cloaking Device" == 1
			"credits" == 5000000
			"flagship planet: Earth"
		# The Diplomacy offer is open in the planet screen; its first button turns Diplomacy on.
		assert
			"diplomacy: active" == 0
		apply
			"test: tries" = 0
		label "accepting"
		input
		input
			key 1
		branch "accepted"
			"diplomacy: active" == 1
		# A landing conversation (the campaign's random post-war reactions) may
		# be in the way: Return goes on with it; then open the commission (K).
		input
			key "return"
		input
		input
			key k
		input
		input
			key 1
		label "accepted"
		assert
			"diplomacy: active" == 1
			"diplomacy: member" == 2
			"diplomacy: rank" == 5
			"salary: Free Worlds" == 8900
		# Luna sells the Navy's capital warships (the Diplomacy event the start applies),
		# as the Diplomacy Probe reports on the next arrival.
		assert
			"event: diplomacy: luna navy yard" == 1
			"diplomacy: luna yard" == 1
		# Close the faction panel the commission opened (on Earth, a Republic world:
		# War Room, Service Record, Close), and take off.
		input
			key 3
		input
		input
		call "Depart"
		apply
			"test: steps to wait" = 30
		label "probing"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "probing"
			"test: steps to wait" > 0
		assert
			"test: luna sells Frigate" == 1
			"test: luna sells Cruiser" == 1
			"test: luna sells Carrier" == 1


# The Free Worlds Veteran's first hour: a border crisis between the Free Worlds
# and the Republic over a minor they share, 10 to 20 days after the commission.
test "Diplomacy - Border Crisis"
	status active
	description "A new pilot on the Free Worlds Veteran start accepts the commission; the border crisis is set for 10-20 days later, and when it comes it is between the Free Worlds and their rival, with a minor as its goal."
	sequence
		input
			key l
		input
			key n
		input
			key up
		input
			key s
		# Typed too soon, the name is lost and the conversation waits: type it
		# again until the pilot has landed on Earth (its landing missions offered).
		apply
			"test: naming" = 0
		label "naming"
		input
		input
		input
			key a
		input
			key "return"
		input
			key b
		input
			key "return"
		input
			key "return"
		input
		input
		input
		apply
			"test: naming" = "test: naming" + 1
		branch "naming"
			"Main Plot Completed Date: offered" == 0
			"test: naming" < 10
		# The commission opens in the planet screen once the mods start: press its
		# first button (Accept) until it has been taken.
		apply
			"test: tries" = 0
		label "accepting"
		input
		input
			key 1
		branch "accepted"
			"diplomacy: active" == 1
		# A landing conversation (the campaign's random post-war reactions) may
		# be in the way: Return goes on with it; then open the commission (K).
		input
			key "return"
		input
		input
			key k
		input
		input
			key 1
		label "accepted"
		apply
			"test: tries" = "test: tries" + 1
		branch "accepting"
			"diplomacy: active" == 0
			"test: tries" < 30
		assert
			"diplomacy: active" == 1
			"diplomacy: member" == 2
			"diplomacy: crisis" == 0
		# Close the faction panel (War Room, Service Record, Close).
		input
			key 3
		input
		input
		# The crisis is set for 10 to 20 days after the commission. Waiting for it
		# here would take off and land 10-20 times, and the campaign's landing
		# messages (Barmy Edward's, and others) stop such a loop at random; so the
		# test checks the day it is set for, then brings it forward with the test
		# override "diplomacy: force crisis" and takes off once.
		assert
			"diplomacy: crisis day" >= "diplomacy: activated" + 10
			"diplomacy: crisis day" <= "diplomacy: activated" + 20
		apply
			"diplomacy: force crisis" = 1
		call "Depart"
		apply
			"test: steps to wait" = 30
		label "flying"
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "flying"
			"test: steps to wait" > 0
		assert
			"diplomacy: crisis" > 0
			"diplomacy: crisis minor" > 0
			"diplomacy: wars" >= 1
			"diplomacy: at war 2" == 1
			"diplomacy: war 1 kind" == 1
			"diplomacy: war 1 goal" > 0
