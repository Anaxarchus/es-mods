test "Shoal Yard E2E"
	status active
	description "The Shoal v0.3 through the Dado loader: refine at Breaker's Rest, upgrade an installed stock outfit and a plugin's gun to Mk I at the Breakers' Yard, take off and load the save again."
	sequence
		inject "Shoal Yard E2E Save"
		call "Load First Savegame"
		assert
			"mod loaders" >= 1
			"mods loaded" >= 1
			"shoal: refinery jobs" == 0
			"outfit (cargo): Shoal Glass" == 25
			"outfit (storage): Shoal Glass" == 10
			"outfit (cargo): Warden Core" == 7
			"outfit (flagship installed): D14-RN Shield Generator" == 1
			"flagship attribute: shield generation" == 200
		# The planet button opens the yard; 1 and 2 press its refinery buttons.
		input
			key k
		input
			key 1
		assert
			"outfit (cargo): Refined Shoal Glass" == 3
			"outfit (cargo): Shoal Glass" == 0
			"outfit (storage): Shoal Glass" == 5
		input
			key 2
		assert
			"outfit (cargo): Refined Warden Core" == 2
			"outfit (cargo): Warden Core" == 1
		# The arrow key clicks the first row (the shield generator): the yard
		# asks to confirm, with Cancel chosen; Right chooses OK.
		input
			key Down
		input
			key Right
		input
			key Return
		assert
			"outfit (flagship installed): D14-RN Shield Generator" == 0
			"outfit (flagship installed): D14-RN Shield Generator (Breakers Mk I)" == 1
			"flagship attribute: shield generation" > 210
			"flagship attribute: shield generation" < 230
			"outfit (cargo): Refined Shoal Glass" == 1
			"credits" == 999000
		input
			key escape
		# Taking off saves the pilot.
		call "Depart"
		assert
			"outfit (flagship installed): D14-RN Shield Generator (Breakers Mk I)" == 1
			"flagship attribute: shield generation" > 210
		input
			key escape
		call "Load First Savegame"
		assert
			"outfit (flagship installed): D14-RN Shield Generator" == 0
			"outfit (flagship installed): D14-RN Shield Generator (Breakers Mk I)" == 1
			"outfit (cargo): Refined Shoal Glass" == 1
			"outfit (cargo): Refined Warden Core" == 2
			"outfit (storage): Shoal Glass" == 5
			"outfit (flagship installed): E2E Sleetgun" == 1
		# Mod API 12: the plugin's gun. The rows are by name: the shield
		# generator's Mk I, then the gun. Down clicks the first row, whose
		# Mk II needs more glass than there is (the yard says so, no confirm,
		# and the row stays chosen); Down clicks the next row, the gun; Right
		# and Return: OK. It costs 2 Refined Shoal Glass (the 1 in the hold,
		# then 1 from storage) and 2,000 credits.
		input
			key k
		input
			key Down
		assert
			"outfit (flagship installed): E2E Sleetgun" == 1
			"credits" == 999000
		input
			key Down
		input
			key Right
		input
			key Return
		assert
			"outfit (flagship installed): E2E Sleetgun" == 0
			"outfit (flagship installed): E2E Sleetgun (Breakers Mk I)" == 1
			"outfit (cargo): Refined Shoal Glass" == 0
			"outfit (storage): Refined Shoal Glass" == 0
			"credits" == 997000
		input
			key escape
		call "Depart"
		assert
			"flagship attribute: shield generation" > 210
			"flagship attribute: shield generation" < 230
			"outfit (flagship installed): E2E Sleetgun (Breakers Mk I)" == 1
		input
			key escape
		call "Load First Savegame"
		assert
			"outfit (flagship installed): E2E Sleetgun" == 0
			"outfit (flagship installed): E2E Sleetgun (Breakers Mk I)" == 1
			"outfit (flagship installed): D14-RN Shield Generator (Breakers Mk I)" == 1
			"outfit (cargo): Refined Shoal Glass" == 0
			"credits" == 997000

test "Shoal E2E Takeoff"
	status partial
	description "Depart (pressing the depart key again every 100 frames) until the flagship is in space."
	sequence
		apply
			"test: takeoff frames" = 0
		label press
		call "Depart"
		label wait
		apply
			"test: takeoff frames" += 1
		branch done
			"flagship landed" == 0
		branch press
			"test: takeoff frames" % 100 == 0
			"test: takeoff frames" < 1000
		branch wait
			"test: takeoff frames" < 1000
		label done
		assert
			"flagship landed" == 0


test "Shoal E2E Land"

test "Shoal Deep E2E"
	status active
	description "The Shoal 0.5.0 through the Dado loader: entering the Deep through the Maw starts a run, landing at Breakers' Haven keeps it, a jump deeper sets the depth, and coming back up through the Upwell ends and counts it."
	sequence
		inject "Shoal Deep E2E Save"
		call "Load First Savegame"
		assert
			"mod loaders" >= 1
			"mods loaded" >= 1
			"shoal: in run" == 0
			"shoal: runs" == 0
		call "Shoal E2E Takeoff"
		assert
			"flagship system: ~The Shoal" == 1
			"shoal: in run" == 0
		call "Shoal E2E Through The Maw"
		assert
			"shoal: in run" == 1
			"shoal: depth" == 0
		navigate
			"travel destination" "Breakers' Haven"
		call "Shoal E2E Land"
		assert
			"flagship planet: Breakers' Haven" == 1
			"shoal: in run" == 1
		call "Shoal E2E Takeoff"
		assert
			"flagship system: ~~Shoal The Sounding" == 1
			"shoal: in run" == 1
		navigate
			travel "~~Shoal Kelpwater"
		apply
			"test: jump frames" = 0
		label jumpdown
		input
			command jump
		label jumpingdown
		apply
			"test: jump frames" += 1
		branch down
			"flagship system: ~~Shoal Kelpwater" == 1
		branch jumpdown
			"test: jump frames" % 200 == 0
			"test: jump frames" < 6000
		branch jumpingdown
			"test: jump frames" < 6000
		label down
		assert
			"flagship system: ~~Shoal Kelpwater" == 1
			"shoal: in run" == 1
			"shoal: depth" == 1
			"shoal: deepest depth" == 1
		navigate
			travel "~~Shoal The Sounding"
		apply
			"test: jump frames" = 0
		label jumpup
		input
			command jump
		label jumpingup
		apply
			"test: jump frames" += 1
		branch up
			"flagship system: ~~Shoal The Sounding" == 1
		branch jumpup
			"test: jump frames" % 200 == 0
			"test: jump frames" < 6000
		branch jumpingup
			"test: jump frames" < 6000
		label up
		assert
			"shoal: in run" == 1
			"shoal: depth" == 0
		call "Shoal E2E Up The Upwell"
		assert
			"flagship system: ~The Shoal" == 1
			"shoal: in run" == 0
			"shoal: runs" == 1
			"shoal: depth" == 0
			"shoal: deepest depth" == 1
