test "Shoal Wait For Takeoff"
	status partial
	description "Depart (pressing the depart key again every 100 frames) and wait until the flagship is in space."
	sequence
		apply
			"test: takeoff frames" = 0
		label press
		call "Depart"
		label takeoff
		apply
			"test: takeoff frames" += 1
		branch done
			"flagship landed" == 0
		branch press
			"test: takeoff frames" % 100 == 0
			"test: takeoff frames" < 1000
		branch takeoff
			"test: takeoff frames" < 1000
		label done
		assert
			"flagship landed" == 0


test "Shoal Land"
	status partial
	description "Land on the travel destination (or nearest planet), giving up after 5000 frames."

		apply
			"test: land frames" = 0
